**Goal:** Set up the repo, a tested plain-C++ `FaithCore` library with the collision layer (`Vec3`, `Aabb`, `SweepHit`, `World`, `BoxWorld`) ported from faith-runner, a golden-trace pipeline that proves the port matches the Rust original, a one-command test script, and an empty UE 5.7 C++ project.

**Tech Stack:** C++20, CMake (the copy bundled with VS 2022), MSVC, doctest 2.5.3 (via FetchContent), Rust stable with cargo, Unreal Engine 5.7, Git and Git LFS, PowerShell.

**Spec:** `docs/superpowers/specs/2026-10-09-faith-movement-port-design.md`

## Global Constraints

- C++ core: C++20, no Unreal headers, no dependencies beyond the standard library (doctest is test-only).
- Core units: metres, **Y up**, matching `faith_move`. No unit conversion inside FaithCore.
- Port fidelity: keep faith-runner's names and comments (Rust `sweep_aabb` becomes C++ `SweepAabb`, and so on), and keep its constants exactly: `TOUCH = 1e-4f`, the `1e-9f` zero-delta check, and the `0.01f` candidate growth.
- Floats are `float` (f32), as in the Rust code. Golden tolerance is `1e-5`.
- Credits: faith-runner (tnrjns) is credited in `CREDITS.md` from the first commit. Nothing from Mirror's Edge's files goes into the repo.
- Every commit message ends with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

## Review Focus

1. **A box starting exactly touching a solid** (`TOUCH` boundary): a sideways move must pass, and a move into the face must hit at `t = 0`. Pinned in Task 3.
2. **A zero component in `delta`** (pure horizontal or vertical moves, the most common case): must take the `1e-9` branch, not divide by zero. Pinned in Task 3.
3. **A box starting inside a solid** must not report a hit against it. Pinned in Task 3.
4. **The golden CSV missing or stale** (the path is wrong, or it was never generated): the test must fail loudly, not pass with zero cases. Pinned in Task 4.
5. **A sweep with zero delta overall:** must return no hit. Pinned in Task 3.

```cpp
#pragma once
#include <algorithm>

namespace faith {

// The subset of glam::Vec3 that faith_move uses. Metres, Y up.
struct Vec3 {
    float x = 0, y = 0, z = 0;

    float& operator[](int i) { return i == 0 ? x : (i == 1 ? y : z); }
    float operator[](int i) const { return i == 0 ? x : (i == 1 ? y : z); }

    friend Vec3 operator+(Vec3 a, Vec3 b) { return {a.x + b.x, a.y + b.y, a.z + b.z}; }
    friend Vec3 operator-(Vec3 a, Vec3 b) { return {a.x - b.x, a.y - b.y, a.z - b.z}; }
    friend Vec3 operator*(Vec3 a, float s) { return {a.x * s, a.y * s, a.z * s}; }
    friend bool operator==(Vec3 a, Vec3 b) = default;

    static Vec3 Splat(float v) { return {v, v, v}; }
    static Vec3 Min(Vec3 a, Vec3 b) { return {std::min(a.x, b.x), std::min(a.y, b.y), std::min(a.z, b.z)}; }
    static Vec3 Max(Vec3 a, Vec3 b) { return {std::max(a.x, b.x), std::max(a.y, b.y), std::max(a.z, b.z)}; }
    Vec3 Clamp(Vec3 lo, Vec3 hi) const { return Max(lo, Min(hi, *this)); }
};

} // namespace faith
```

```cpp
#pragma once
#include <optional>
#include "faith/Vec3.h"

namespace faith {

// Port of faith_move::world. Everything the controller knows about the level goes through
// World: sweep the player's box through it (first contact, with the surface normal) and ask
// whether a box is clear. Those are the two questions an engine's physics answers natively.

struct Aabb {
    Vec3 min, max;

    static Aabb New(Vec3 a, Vec3 b) { return {Vec3::Min(a, b), Vec3::Max(a, b)}; }

    /// Strict overlap (touching faces do not count).
    bool Overlaps(const Aabb& o) const {
        constexpr float E = 1e-4f;
        return min.x < o.max.x - E && max.x > o.min.x + E
            && min.y < o.max.y - E && max.y > o.min.y + E
            && min.z < o.max.z - E && max.z > o.min.z + E;
    }
    Aabb Translated(Vec3 d) const { return {min + d, max + d}; }
    Aabb Union(const Aabb& o) const { return {Vec3::Min(min, o.min), Vec3::Max(max, o.max)}; }
};

/// Where a swept box first touched something.
struct SweepHit {
    float t;      ///< How much of the move was made before the contact (0..1).
    Vec3 normal;  ///< The surface's normal at the contact, facing back at the box.
    Vec3 point;   ///< The contact point, on the surface.
};

/// The level, as the controller sees it.
class World {
public:
    virtual ~World() = default;
    /// Sweep a box (half size `half`, centred at `start`) along `delta` and return the first
    /// solid it runs into before the end. Solids it starts inside (beyond touching) don't
    /// count, like the game's extent traces.
    virtual std::optional<SweepHit> Sweep(Vec3 half, Vec3 start, Vec3 delta) const = 0;
    /// Whether anything solid is inside `region` (touching its faces doesn't count).
    virtual bool Overlaps(const Aabb& region) const = 0;
    bool IsFree(const Aabb& region) const { return !Overlaps(region); }
};

} // namespace faith
```

- [ ] **Step 5: Write `src/BoxWorld.cpp`, a line-for-line port**

```cpp
#include "faith/BoxWorld.h"
#include <cmath>
#include <limits>

namespace faith {
namespace {

/// How far a box may be inside something and still only be touching it.
constexpr float TOUCH = 1e-4f;

struct AxisHit { float t; Vec3 normal; };

/// Sweep a box against one axis-aligned box: (fraction, normal) of the first contact. The
/// usual slab test on the box grown by `half`, with the overlap judged strictly (TOUCH): a
/// box only touching sideways is passed, one touching ahead is hit at once.
std::optional<AxisHit> SweepAabb(const Aabb& b, Vec3 half, Vec3 c, Vec3 delta) {
    const Vec3 lo = b.min - half;
    const Vec3 hi = b.max + half;
    float enterS = -std::numeric_limits<float>::infinity();
    float exitS = std::numeric_limits<float>::infinity();
    float enter = 0.0f;
    Vec3 normal{};
    for (int a = 0; a < 3; ++a) {
        const float d = delta[a];
        if (std::fabs(d) < 1e-9f) {
            if (c[a] <= lo[a] + TOUCH || c[a] >= hi[a] - TOUCH) return std::nullopt;
            continue;
        }
        // Strict slab (shrunk by TOUCH) decides whether the paths overlap; the true faces
        // say when the contact is.
        const float sa = (lo[a] + TOUCH - c[a]) / d;
        const float sb = (hi[a] - TOUCH - c[a]) / d;
        const float se = sa < sb ? sa : sb;
        const float sx = sa < sb ? sb : sa;
        const float face = d > 0.0f ? (lo[a] - c[a]) / d : (hi[a] - c[a]) / d;
        if (se > enterS) {
            enterS = se;
            enter = face;
            normal = Vec3{};
            normal[a] = d > 0.0f ? -1.0f : 1.0f;   // -d.signum()
        }
        exitS = std::min(exitS, sx);
    }
    if (normal == Vec3{} || enterS >= exitS || enterS >= 1.0f || exitS <= 0.0f || enterS <= 0.0f)
        return std::nullopt;
    const float t = std::max(enter, 0.0f);
    if (t < 1.0f) return AxisHit{t, normal};
    return std::nullopt;
}

Vec3 ContactOnBox(const Aabb& b, Vec3 half, Vec3 c, Vec3 delta, float t, Vec3 n) {
    const Vec3 at = c + delta * t;
    Vec3 p = at.Clamp(b.min, b.max);
    for (int a = 0; a < 3; ++a)
        if (n[a] != 0.0f) p[a] = at[a] - n[a] * half[a];
    return p;
}

/// The first contact of a box swept through some boxes.
std::optional<SweepHit> SweepBoxes(const std::vector<Aabb>& boxes, Vec3 half, Vec3 start, Vec3 delta) {
    const Aabb from = Aabb::New(start - half, start + half);
    std::optional<SweepHit> best;
    const Aabb* bestBox = nullptr;
    for (const Aabb& b : boxes) {
        if (b.Overlaps(from)) continue;
        if (auto h = SweepAabb(b, half, start, delta)) {
            if (!best || h->t < best->t) { best = SweepHit{h->t, h->normal, {}}; bestBox = &b; }
        }
    }
    if (best) best->point = ContactOnBox(*bestBox, half, start, delta, best->t, best->normal);
    return best;
}

} // namespace

std::optional<SweepHit> BoxWorld::Sweep(Vec3 half, Vec3 start, Vec3 delta) const {
    const Aabb a = Aabb::New(start - half, start + half);
    const Aabb region = a.Union(a.Translated(delta));
    // Candidates touching the swept region (with a hair, for touching faces).
    const Aabb grown = Aabb::New(region.min - Vec3::Splat(0.01f), region.max + Vec3::Splat(0.01f));
    std::vector<Aabb> candidates;
    for (const Aabb& b : boxes)
        if (b.Overlaps(grown)) candidates.push_back(b);
    return SweepBoxes(candidates, half, start, delta);
}

bool BoxWorld::Overlaps(const Aabb& region) const {
    for (const Aabb& b : boxes)
        if (b.Overlaps(region)) return true;
    return false;
}

} // namespace faith
```

**Interfaces:**

- Produces: the formats used by every later golden file.
  - `scene.csv`: a header line, then `minx,miny,minz,maxx,maxy,maxz`, one box per line. The scene is defined once, in Rust, so it can't drift between the two languages.
  - `sweeps.csv`: a header line, then `hx,hy,hz,sx,sy,sz,dx,dy,dz,hit,t,nx,ny,nz,px,py,pz`, one case per line. `hit` is 0 or 1; when it's 0, the remaining fields are 0.

Why two kinds of cases: random floats almost never land exactly on a touching position. A review found that six different broken ports (TOUCH handling, the strict tie rule, the start-inside skip, and others) all produce **identical** output on purely random cases. So the tool also writes **structured** cases: starts exactly on, and 1e-5 either side of, every face; moves along, into and away from faces; and diagonal approaches to corners. These are what make the golden test check the boundary behaviour, not just the arithmetic.

```rust
//! Writes reference results from faith-runner's faith_move for FaithCore's golden tests.
use faith_move::world::{Aabb, BoxWorld, World};
use glam::Vec3;
use std::fmt::Write as _;

/// The golden scene. The floor is two flush boxes, so seams get tested too.
const SCENE: [([f32; 3], [f32; 3]); 6] = [
    ([-10.0, -1.0, -10.0], [0.0, 0.0, 10.0]),  // floor, west half
    ([0.0, -1.0, -10.0], [10.0, 0.0, 10.0]),   // floor, east half (seam at x = 0)
    ([2.0, 0.0, -1.0], [3.0, 3.0, 1.0]),       // wall
    ([-4.0, 0.0, 2.0], [-2.0, 1.2, 4.0]),      // waist-high block
    ([-1.0, 2.5, -6.0], [1.0, 2.7, -3.0]),     // overhead pipe
    ([4.0, 0.0, 4.0], [4.5, 6.0, 4.5]),        // pillar
];

/// Deterministic LCG, so the file only changes when the reference does.
struct Lcg(u64);
impl Lcg {
    fn next(&mut self) -> f32 {
        self.0 = self.0.wrapping_mul(6364136223846793005).wrapping_add(1442695040888963407);
        ((self.0 >> 40) as f32) / ((1u64 << 24) as f32)
    }
    fn range(&mut self, lo: f32, hi: f32) -> f32 { lo + (hi - lo) * self.next() }
    /// Snap a third of values to 0 so the zero-delta branch is exercised.
    fn axis(&mut self, m: f32) -> f32 { if self.next() < 0.33 { 0.0 } else { self.range(-m, m) } }
}

// `{}` prints an f32 in the shortest form that round-trips, so C++'s stof reads back the same bits.
fn f(v: Vec3) -> String { format!("{},{},{}", v.x, v.y, v.z) }

fn row(w: &BoxWorld, half: Vec3, start: Vec3, delta: Vec3) -> String {
    match w.sweep(half, start, delta) {
        Some(h) => format!("{},{},{},1,{},{},{}\n", f(half), f(start), f(delta), h.t, f(h.normal), f(h.point)),
        None => format!("{},{},{},0,0,0,0,0,0,0,0\n", f(half), f(start), f(delta)),
    }
}

/// Starts exactly on (and 1e-5 either side of) every face of every box, and diagonal corner
/// approaches: the boundary cases random floats never reach.
fn structured(w: &BoxWorld, out: &mut String) {
    let halves = [Vec3::new(0.3, 0.9, 0.3), Vec3::new(0.5, 0.9, 0.5)];
    let offs = [0.0f32, 1e-5, -1e-5, 0.25];
    for (lo, hi) in SCENE {
        let (lo, hi) = (Vec3::from(lo), Vec3::from(hi));
        let mid = (lo + hi) * 0.5;
        for half in halves {
            for a in 0..3 {
                for s in [-1.0f32, 1.0] {
                    for off in offs {
                        // Outside the face on side `s` of axis `a`, `off` away from touching.
                        let mut start = mid;
                        start[a] = if s < 0.0 { lo[a] - half[a] - off } else { hi[a] + half[a] + off };
                        let mut into = Vec3::ZERO;
                        into[a] = -s * 0.5;
                        let mut deltas = vec![Vec3::ZERO, into, -into]; // still, into the face, away
                        for b in 0..3 {
                            if b == a { continue; }
                            for t in [-0.5f32, 0.5] {
                                let mut side = Vec3::ZERO;
                                side[b] = t;
                                deltas.push(side);        // along the face
                                deltas.push(side + into); // diagonally into it
                            }
                        }
                        for d in deltas { out.push_str(&row(w, half, start, d)); }
                    }
                }
            }
            // Corner approaches: from outside the (min x, min z) vertical edge, and over the top edge.
            for off in offs {
                let c = Vec3::new(lo.x - half.x - off, mid.y, lo.z - half.z - off);
                for d in [Vec3::new(1.0, 0.0, 1.0), Vec3::new(1.0, 0.0, 0.5), Vec3::new(0.5, 0.0, 1.0)] {
                    out.push_str(&row(w, half, c, d));
                }
                let top = Vec3::new(lo.x - half.x - off, hi.y + half.y + off, mid.z);
                out.push_str(&row(w, half, top, Vec3::new(1.0, -1.0, 0.0)));
            }
        }
    }
}

fn main() {
    let mut w = BoxWorld::default();
    let mut scene = String::from("minx,miny,minz,maxx,maxy,maxz\n");
    for (a, b) in SCENE {
        w.add(Aabb::new(Vec3::from(a), Vec3::from(b)));
        writeln!(scene, "{},{}", f(Vec3::from(a)), f(Vec3::from(b))).unwrap();
    }

    let mut out = String::from("hx,hy,hz,sx,sy,sz,dx,dy,dz,hit,t,nx,ny,nz,px,py,pz\n");
    structured(&w, &mut out);
    let mut r = Lcg(0x5EED);
    for _ in 0..4000 {
        let half = Vec3::new(r.range(0.1, 0.5), r.range(0.3, 1.0), r.range(0.1, 0.5));
        let start = Vec3::new(r.range(-6.0, 6.0), r.range(0.0, 4.0), r.range(-6.0, 6.0));
        let delta = Vec3::new(r.axis(6.0), r.axis(4.0), r.axis(6.0));
        out.push_str(&row(&w, half, start, delta));
    }

    let dir = concat!(env!("CARGO_MANIFEST_DIR"), "/../../FaithCore/tests/golden");
    std::fs::create_dir_all(dir).unwrap();
    std::fs::write(format!("{dir}/scene.csv"), scene).unwrap();
    std::fs::write(format!("{dir}/sweeps.csv"), out).unwrap();
    println!("wrote {dir}/scene.csv and sweeps.csv");
}
```

```cpp
#include <cmath>

std::vector<std::vector<float>> ReadCsv(const std::string& name) {
    std::ifstream in(std::string(FAITH_GOLDEN_DIR) + "/" + name);
    std::vector<std::vector<float>> rows;

        while (std::getline(ss, cell, ',')) row.push_back(std::stof(cell));

        const bool want = r[9] != 0.0f;

            diff = std::string("hit: want ") + (want ? "1" : "0") + ", got " + (h ? "1" : "0");
        } else if (want) {

            for (int k = 0; k < 7 && diff.empty(); ++k) {
```
